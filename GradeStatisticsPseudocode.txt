BEGIN

SET number of grades to 10
SET total to 0

FOR grade number from 1 through 10

    ASK the user to enter a grade
    READ the grade

    ADD the grade to the total value

    IF this is the first grade
        SET maximum equal to the grade
        SET minimum equal to the grade
    ELSE
        IF the grade is greater than maximum
            SET maximum equal to the grade
        END IF

        IF the grade is less than minimum
            SET minimum equal to the grade
        END IF
    END IF

END FOR

CALCULATE average by dividing total by the number 10

DISPLAY the average grade
DISPLAY the maximum grade
DISPLAY the minimum grade

END
